## Global Constraints

- No new Cargo dependencies; gameplay modules never import wgpu and test on the host (`cd rust && cargo test --lib`).
- Fixed physics step 1/60 s; all timing from the passed `dt`; the game loop caps a frame at 0.1 s (existing).
- Coordinates: x is the car's LEFT, y up, z forward at yaw 0; heading `yaw = atan2(fwd.x, fwd.z)`; the aircraft's right-hand side is `(-cos yaw, 0, sin yaw)`; a positive stick input turns RIGHT (yaw decreases).
- Aircraft numbers (from the spec): heli tilt 25°, speed cap 35 m/s, vertical 6 m/s (accel 4), yaw 1.6 rad/s; drone tilt 40°, cap 28, vertical 10 (accel 12), yaw 3.0; tilt lag 0.25 s; soft landing = vertical speed ≤ 3 m/s and tilt ≤ 20°; tilt > 35° at contact crashes; ceiling 400 m; wall margin `WORLD_LIMIT - 20`.
- Rope rest length 8 m, k = 60, c = 6, cargo mass 1; pick-up: lower than 4 m above the cargo for 1.5 s; delivery: inside the 6 m circle, speed < 2 m/s, cargo within 1 m of the ground; swing damage 4 per second above 12 m/s relative speed; impact damage `(v - 6) * 2.5`.
- Rings: radius heli 8 m, drone 6 m, taken in order by plane crossing; recon capture sphere 40 m with 2 s dwell, any order.
- `time_limit`: rings `1.5 * len / v_ref + 20` (v_ref heli 18, drone 15); recon `1.4 * len / 15 + 25`; sling `1.6 * len / 12 + 35`.
- Profile kind ids: Cargo 0, Race 1, Rings 2, Sling 3, Recon 4 (old profiles with 0/1 keep loading); failure codes in the state array: 1 time, 2 cargo, 3 abandoned, 4 crashed.
- UI strings bilingual via `t(en, ru)`; docs EN+RU. Scenes mode, ground missions and all existing tests keep passing.
- Commit messages end with `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.
- Rulings vs the spec (decided while planning): (a) the sling delivery limit is `1.6 * len / 12 + 35` (the spec left it open; +15 s over the ground formula for the pick-up); (b) the recon zone is drawn as a 40 m ground circle plus a pillar, not a sphere (an opaque sphere would hide the scene); (c) `Mission` gets `aircraft: Option<AircraftKind>` and every test literal gains `aircraft: None`; (d) one `RunInfo` view over `MissionRun` and `AirRun` replaces direct `World.mission` reads in `fill_state` and the recorder; (e) the menu info array grows to 9 floats per mission and 6 missions (58 floats), adding an aircraft code.

```rust
//! Arcade helicopter / drone: tilt-driven acceleration, auto-hover, soft landing vs crash.
use super::math::{v3, Q, V3};
use super::props::{Prop, PropKind};
use super::terrain::Terrain;
use super::world::WORLD_LIMIT;

pub const CEILING: f32 = 400.0;
const G: f32 = 9.81;

#[derive(Clone, Copy, Debug, PartialEq, Eq)] pub enum AircraftKind { Heli, Drone }
#[derive(Clone, Copy, Debug, PartialEq, Eq)] pub enum Status { Flying, Landed, Crashed }

#[derive(Clone, Copy, Debug, Default)]
pub struct FlightInput { pub yaw: f32, pub lift: f32, pub pitch: f32, pub roll: f32 }
impl FlightInput {
    pub fn sanitized(self) -> FlightInput {
        let f = |v: f32| if v.is_finite() { v.clamp(-1.0, 1.0) } else { 0.0 };
        FlightInput { yaw: f(self.yaw), lift: f(self.lift), pitch: f(self.pitch), roll: f(self.roll) }
    }
    /// Any stick beyond the dead zone (starts the mission clock).
    pub fn active(&self) -> bool { [self.yaw, self.lift, self.pitch, self.roll].iter().any(|v| v.is_finite() && v.abs() > 0.05) }
}

#[derive(Clone, Copy, Debug)]
pub struct AircraftSpec { pub max_tilt: f32, pub tilt_lag: f32, pub max_speed: f32, pub drag: f32, pub vmax: f32, pub vacc: f32, pub yaw_rate: f32, pub yaw_lag: f32, pub radius: f32 }
impl AircraftSpec {
    pub fn of(kind: AircraftKind) -> AircraftSpec {
        match kind {
            AircraftKind::Heli => AircraftSpec { max_tilt: 25f32.to_radians(), tilt_lag: 0.25, max_speed: 35.0, drag: 0.13, vmax: 6.0, vacc: 4.0, yaw_rate: 1.6, yaw_lag: 0.3, radius: 1.8 },
            AircraftKind::Drone => AircraftSpec { max_tilt: 40f32.to_radians(), tilt_lag: 0.25, max_speed: 28.0, drag: 0.29, vmax: 10.0, vacc: 12.0, yaw_rate: 3.0, yaw_lag: 0.15, radius: 0.7 },
        }
    }
}

#[derive(Clone, Copy, Debug)]
pub struct Aircraft {
    pub kind: AircraftKind, pub spec: AircraftSpec,
    pub pos: V3, pub vel: V3,
    /// Heading angle: forward = (sin yaw, 0, cos yaw). Pitch is nose-down positive, roll is right-down positive.
    pub yaw: f32, pub pitch: f32, pub roll: f32, pub yaw_rate: f32,
    pub status: Status,
}

impl Aircraft {
    pub fn new(kind: AircraftKind, pos: V3, yaw: f32) -> Aircraft {
        Aircraft { kind, spec: AircraftSpec::of(kind), pos, vel: V3::ZERO, yaw, pitch: 0.0, roll: 0.0, yaw_rate: 0.0, status: Status::Flying }
    }
    pub fn forward(&self) -> V3 { v3(self.yaw.sin(), 0.0, self.yaw.cos()) }
    pub fn right(&self) -> V3 { v3(-self.yaw.cos(), 0.0, self.yaw.sin()) }
    pub fn speed(&self) -> f32 { self.vel.len() }
    pub fn rotation(&self) -> Q {
        Q::axis_angle(v3(0., 1., 0.), self.yaw).mul(Q::axis_angle(v3(1., 0., 0.), self.pitch)).mul(Q::axis_angle(v3(0., 0., 1.), self.roll))
    }

    pub fn step(&mut self, t: &Terrain, props: &[Prop], inp: &FlightInput, dt: f32) {
        if self.status == Status::Crashed { return; }
        let inp = inp.sanitized();
        let s = self.spec;
        let k = |lag: f32| 1.0 - (-dt / lag).exp();
        // heading: a positive input turns right, which decreases the heading angle
        let target_rate = -inp.yaw * s.yaw_rate;
        self.yaw_rate += (target_rate - self.yaw_rate) * k(s.yaw_lag);
        self.yaw += self.yaw_rate * dt;
        // tilt
        self.pitch += (inp.pitch * s.max_tilt - self.pitch) * k(s.tilt_lag);
        self.roll += (inp.roll * s.max_tilt - self.roll) * k(s.tilt_lag);
        // horizontal motion
        let (f, r) = (self.forward(), self.right());
        let acc = f * (G * self.pitch.tan()) + r * (G * self.roll.tan());
        let mut vh = v3(self.vel.x, 0.0, self.vel.z);
        vh = vh + acc * dt - vh * (s.drag * dt);
        let sp = vh.len();
        if sp > s.max_speed { vh = vh * (s.max_speed / sp); }
        // vertical motion
        let vt = inp.lift * s.vmax;
        let mut vy = self.vel.y + (vt - self.vel.y).clamp(-s.vacc * dt, s.vacc * dt);
        if self.pos.y >= CEILING && vy > 0.0 { vy = 0.0; }
        self.vel = v3(vh.x, vy, vh.z);
        self.pos = self.pos + self.vel * dt;
        // world wall
        let lim = WORLD_LIMIT - 20.0;
        if self.pos.x.abs() > lim { self.pos.x = self.pos.x.clamp(-lim, lim); if self.pos.x * self.vel.x > 0.0 { self.vel.x = 0.0; } }
        if self.pos.z.abs() > lim { self.pos.z = self.pos.z.clamp(-lim, lim); if self.pos.z * self.vel.z > 0.0 { self.vel.z = 0.0; } }
        // props
        for p in props {
            let hit = match p.kind {
                PropKind::Rock => (self.pos - (p.pos + v3(0., p.radius * 0.5, 0.))).len() < p.radius + s.radius,
                PropKind::Tree => self.pos.y < p.pos.y + p.height && (self.pos.x - p.pos.x).hypot(self.pos.z - p.pos.z) < p.radius + s.radius,
            };
            if hit { self.status = Status::Crashed; self.vel = V3::ZERO; return; }
        }
        // ground
        let g = t.height(self.pos.x, self.pos.z);
        if self.pos.y - g < 0.3 {
            let tilt = self.pitch.abs().max(self.roll.abs());
            let touchdown_speed = -self.vel.y;
            if touchdown_speed > 3.0 || tilt > 20f32.to_radians() {
                self.status = Status::Crashed; self.vel = V3::ZERO; return;
            }
            self.pos.y = g + 0.3;
            if self.vel.y < 0.0 { self.vel.y = 0.0; }
            let damp = (1.0 - 4.0 * dt).max(0.0);
            self.vel.x *= damp; self.vel.z *= damp;
            self.status = Status::Landed;
        } else if self.status == Status::Landed && self.vel.y > 0.0 || self.pos.y - g > 0.5 {
            self.status = Status::Flying;
        }
    }
}
```

```rust
//! Cargo on a rope under the helicopter: a point mass on a one-sided spring-damper (taut only beyond the rest length).
use super::math::{v3, V3};
use super::terrain::Terrain;

pub const ROPE: f32 = 8.0;
const K: f32 = 60.0;
const C: f32 = 6.0;
const DRAG: f32 = 0.8;
const SUB: usize = 4;

#[derive(Clone, Copy, Debug)]
pub struct Sling { pub pos: V3, pub vel: V3, pub attached: bool }

impl Sling {
    pub fn on_ground(pos: V3) -> Sling { Sling { pos, vel: V3::ZERO, attached: false } }
    pub fn attach(&mut self) { self.attached = true; }
    pub fn relative_speed(&self, attach_vel: V3) -> f32 { (self.vel - attach_vel).len() }
    pub fn rope_taut(&self, attach: V3) -> bool { (attach - self.pos).len() >= ROPE - 0.05 }

    /// Advances the cargo; returns the speed of a real ground impact (> 0.5 m/s into the ground) in this step.
    pub fn step(&mut self, attach: V3, attach_vel: V3, t: &Terrain, dt: f32) -> Option<f32> {
        let h = dt / SUB as f32;
        let mut impact: Option<f32> = None;
        for _ in 0..SUB {
            let mut force = v3(0., -9.81, 0.) - self.vel * DRAG;
            if self.attached {
                let d = attach - self.pos; let dist = d.len();
                if dist > ROPE && dist > 1e-4 {
                    let n = d * (1.0 / dist);
                    let f = (K * (dist - ROPE) + C * (attach_vel - self.vel).dot(n)).max(0.0);
                    force = force + n * f;
                }
            }
            self.vel = self.vel + force * h;
            self.pos = self.pos + self.vel * h;
            let g = t.height(self.pos.x, self.pos.z) + 0.3;
            if self.pos.y < g {
                if self.vel.y < -0.5 { let v = self.vel.len(); impact = Some(impact.map_or(v, |p: f32| p.max(v))); }
                self.pos.y = g;
                if self.vel.y < 0.0 { self.vel.y = 0.0; }
                self.vel.x *= 0.8; self.vel.z *= 0.8;
            }
        }
        // hard cap: a teleported attach point must not stretch the rope without bound
        if self.attached {
            let d = attach - self.pos; let dist = d.len();
            if dist > ROPE * 1.5 { self.pos = attach - d * (1.0 / dist) * (ROPE * 1.5); self.vel = attach_vel; }
        }
        impact
    }
}
```

```rust
//! Point generation for air missions (rings, sling pads, recon zones).
use super::math::{v3, V3};
use super::missions::{at, nearest_ok};
use super::terrain::{hash01, Terrain};
use super::world::WORLD_LIMIT;

fn clampw(v: f32) -> f32 { v.clamp(-(WORLD_LIMIT - 40.0), WORLD_LIMIT - 40.0) }

/// 5-8 rings along a 500-1400 m path with a zigzag of at most 80 m; heights above ground 15..90 (drone ..120).
pub fn rings_points(t: &Terrain, key: u64, drone: bool) -> Vec<V3> {
    let len = 500.0 + 900.0 * hash01(key, 31, 1);
    let a = hash01(key, 31, 2) * std::f32::consts::TAU;
    let dir = v3(a.cos(), 0.0, a.sin()); let side = v3(-dir.z, 0.0, dir.x);
    let n = 5 + (hash01(key, 31, 3) * 4.0) as usize % 4;
    let hmax = if drone { 120.0 } else { 90.0 };
    (1..=n).map(|i| {
        let k = i as f32 / n as f32;
        let zig = if i % 2 == 0 { 1.0 } else { -1.0 } * 80.0 * hash01(key, 100 + i as i32, 4);
        let (x, z) = (clampw(dir.x * len * k + side.x * zig), clampw(dir.z * len * k + side.z * zig));
        v3(x, t.height(x, z) + 15.0 + (hmax - 15.0) * hash01(key, 200 + i as i32, 5), z)
    }).collect()
}

/// Pick-up pad A (80-200 m from the start) and drop pad B (400-1000 m from A), both on gentle ground.
pub fn sling_points(t: &Terrain, key: u64) -> Vec<V3> {
    let a1 = hash01(key, 41, 1) * std::f32::consts::TAU; let d1 = 80.0 + 120.0 * hash01(key, 41, 2);
    let pa = nearest_ok(t, a1.cos() * d1, a1.sin() * d1);
    let a2 = hash01(key, 41, 3) * std::f32::consts::TAU; let d2 = 400.0 + 600.0 * hash01(key, 41, 4);
    let pb = nearest_ok(t, clampw(pa.x + a2.cos() * d2), clampw(pa.z + a2.sin() * d2));
    vec![pa, pb]
}

/// 3-5 capture points at 20..100 m height around a loop of 600-1200 m that passes the start.
pub fn recon_points(t: &Terrain, key: u64) -> Vec<V3> {
    let len = 600.0 + 600.0 * hash01(key, 51, 1);
    let r = len / std::f32::consts::TAU;
    let a = hash01(key, 51, 2) * std::f32::consts::TAU;
    let dir = v3(a.cos(), 0.0, a.sin());
    let centre = dir * r;
    let n = 3 + (hash01(key, 51, 3) * 3.0) as usize % 3;
    (1..=n).map(|i| {
        let th = a + std::f32::consts::PI + i as f32 * std::f32::consts::TAU / (n as f32 + 1.0);
        let (x, z) = (clampw(centre.x + r * th.cos()), clampw(centre.z + r * th.sin()));
        v3(x, t.height(x, z) + 20.0 + 80.0 * hash01(key, 300 + i as i32, 6), z)
    }).collect()
}
```

```rust
//! State machine of one air mission (rings, sling delivery, recon). All time comes from `dt`.
use super::flight::{Aircraft, AircraftKind, Status};
use super::math::V3;
use super::medals::{grade_cargo, grade_race};
use super::missions::{ring_normal, Mission, MissionKind, RING_RADIUS_DRONE, RING_RADIUS_HELI, ZONE_RADIUS};
use super::run::{Failure, Outcome, Phase};
use super::sling::Sling;

#[derive(Clone, Copy, Debug, PartialEq, Eq)] pub enum AirAction { AttachCargo }

pub struct AirRun {
    m: Mission, phase: Phase, time: f32, outcome: Option<Outcome>, integrity: f32,
    next: usize, captured: Vec<bool>, zone: Option<usize>, dwell: f32,
    attached: bool, hover: f32, prev: Option<V3>,
}

impl AirRun {
    pub fn new(m: Mission) -> AirRun {
        let n = m.checkpoints.len();
        AirRun { m, phase: Phase::Ready, time: 0.0, outcome: None, integrity: 100.0, next: 0, captured: vec![false; n], zone: None, dwell: 0.0, attached: false, hover: 0.0, prev: None }
    }
    pub fn phase(&self) -> Phase { self.phase }
    pub fn outcome(&self) -> Option<Outcome> { self.outcome }
    pub fn time(&self) -> f32 { self.time }
    pub fn time_left(&self) -> f32 { (self.m.time_limit - self.time).max(0.0) }
    pub fn integrity(&self) -> f32 { self.integrity }
    pub fn mission(&self) -> &Mission { &self.m }
    pub fn total(&self) -> usize { if self.m.kind == MissionKind::Sling { 2 } else { self.m.checkpoints.len() } }
    pub fn next_index(&self) -> usize {
        match self.m.kind { MissionKind::Recon => self.captured.iter().filter(|c| **c).count(), MissionKind::Sling => self.attached as usize, _ => self.next }
    }
    pub fn target(&self, from: V3) -> Option<V3> {
        if self.phase == Phase::Done { return None; }
        match self.m.kind {
            MissionKind::Rings => self.m.checkpoints.get(self.next).copied(),
            MissionKind::Sling => self.m.checkpoints.get(self.attached as usize).copied(),
            _ => self.m.checkpoints.iter().zip(&self.captured).filter(|(_, c)| !**c).map(|(p, _)| *p)
                .min_by(|a, b| (*a - from).len().partial_cmp(&(*b - from).len()).unwrap()),
        }
    }
    pub fn abandon(&mut self) { if self.phase != Phase::Done { self.finish(Outcome::Failed(Failure::Abandoned)); } }
    fn finish(&mut self, o: Outcome) { self.phase = Phase::Done; self.outcome = Some(o); }
    fn pass(&mut self) {
        let medal = if self.m.kind == MissionKind::Sling { grade_cargo(self.time, self.m.time_limit, self.integrity) } else { grade_race(self.time, self.m.time_limit) };
        self.finish(Outcome::Passed { medal, time: self.time, integrity: self.integrity.max(0.0) });
    }

    pub fn update(&mut self, a: &Aircraft, sling: Option<&Sling>, impact: Option<f32>, active: bool, dt: f32) -> Option<AirAction> {
        if self.phase == Phase::Done { return None; }
        if self.phase == Phase::Ready {
            if active { self.phase = Phase::Running; } else { self.prev = Some(a.pos); return None; }
        }
        self.time += dt;
        let prev = self.prev.unwrap_or(a.pos);
        self.prev = Some(a.pos);
        if a.status == Status::Crashed { self.finish(Outcome::Failed(Failure::Crashed)); return None; }
        let mut action = None;
        match self.m.kind {
            MissionKind::Rings => {
                if let Some(p) = self.m.checkpoints.get(self.next).copied() {
                    let n = ring_normal(&self.m, self.next);
                    let (d0, d1) = ((prev - p).dot(n), (a.pos - p).dot(n));
                    if d0 < 0.0 && d1 >= 0.0 {
                        let q = prev + (a.pos - prev) * (-d0 / (d1 - d0));
                        let r = if a.kind == AircraftKind::Drone { RING_RADIUS_DRONE } else { RING_RADIUS_HELI };
                        if (q - p).len() <= r { self.next += 1; if self.next >= self.m.checkpoints.len() { self.pass(); return None; } }
                    }
                }
            }
            MissionKind::Recon => {
                let inside = self.m.checkpoints.iter().enumerate().find(|(i, p)| !self.captured[*i] && (a.pos - **p).len() <= ZONE_RADIUS).map(|(i, _)| i);
                match inside {
                    Some(i) if self.zone == Some(i) => self.dwell += dt,
                    Some(i) => { self.zone = Some(i); self.dwell = dt; }
                    None => { self.zone = None; self.dwell = 0.0; }
                }
                if let (Some(i), true) = (self.zone, self.dwell >= 2.0) { self.captured[i] = true; self.zone = None; self.dwell = 0.0; }
                if self.captured.iter().all(|c| *c) { self.pass(); return None; }
            }
            MissionKind::Sling => {
                if let Some(s) = sling {
                    if !self.attached {
                        let above = a.pos.y - s.pos.y;
                        let close = (a.pos.x - s.pos.x).hypot(a.pos.z - s.pos.z) < 3.0 && above < 4.0 && above > 0.0;
                        self.hover = if close { self.hover + dt } else { 0.0 };
                        if self.hover >= 1.5 { self.attached = true; action = Some(AirAction::AttachCargo); }
                    } else {
                        if let Some(v) = impact { if v > 6.0 { self.integrity -= (v - 6.0) * 2.5; } }
                        if s.relative_speed(a.vel) > 12.0 { self.integrity -= 4.0 * dt; }
                        let b = self.m.checkpoints[1];
                        let in_circle = (s.pos.x - b.x).hypot(s.pos.z - b.z) <= 6.0 && s.pos.y - b.y < 1.0 && s.vel.len() < 2.0;
                        if self.integrity > 0.0 && in_circle { self.pass(); return None; }
                        if self.integrity <= 0.0 { self.integrity = 0.0; self.finish(Outcome::Failed(Failure::CargoDestroyed)); return None; }
                    }
                }
            }
            _ => {}
        }
        if self.time > self.m.time_limit { self.finish(Outcome::Failed(Failure::TimeUp)); }
        action
    }
}
```
